1. Get the Orchestrator URL by running these commands:
{{- if .Values.ingress.enabled }}
{{- range $host := .Values.ingress.hosts }}
  {{- range .paths }}
  http{{ if $.Values.ingress.tls }}s{{ end }}://{{ $host.host }}{{ .path }}
  {{- end }}
{{- end }}
{{- else if contains "NodePort" .Values.service.type }}
  export NODE_PORT=$(kubectl get --namespace {{ include "orchestrator.namespace" . }} -o jsonpath="{.spec.ports[0].nodePort}" services {{ include "orchestrator.fullname" . }})
  export NODE_IP=$(kubectl get nodes --namespace {{ include "orchestrator.namespace" . }} -o jsonpath="{.items[0].status.addresses[0].address}")
  echo http://$NODE_IP:$NODE_PORT
{{- else if contains "LoadBalancer" .Values.service.type }}
     NOTE: It may take a few minutes for the LoadBalancer IP to be available.
           You can watch the status of by running 'kubectl get --namespace {{ include "orchestrator.namespace" . }} svc -w {{ include "orchestrator.fullname" . }}'
  export SERVICE_IP=$(kubectl get svc --namespace {{ include "orchestrator.namespace" . }} {{ include "orchestrator.fullname" . }} --template "{{"{{ range (index .status.loadBalancer.ingress 0) }}{{.}}{{ end }}"}}")
  echo http://$SERVICE_IP:{{ .Values.service.port }}
{{- else if contains "ClusterIP" .Values.service.type }}
  export POD_NAME=$(kubectl get pods --namespace {{ include "orchestrator.namespace" . }} -l "app.kubernetes.io/name={{ include "orchestrator.name" . }},app.kubernetes.io/instance={{ .Release.Name }}" -o jsonpath="{.items[0].metadata.name}")
  export CONTAINER_PORT=$(kubectl get pod --namespace {{ include "orchestrator.namespace" . }} $POD_NAME -o jsonpath="{.spec.containers[0].ports[0].containerPort}")
  echo "Visit http://127.0.0.1:8080 to use your application"
  kubectl --namespace {{ include "orchestrator.namespace" . }} port-forward $POD_NAME 8080:$CONTAINER_PORT
{{- end }}
{{- if .Values.namespaceOverride }}

NOTE: namespaceOverride is set to "{{ include "orchestrator.namespace" . }}". Ensure this namespace
      pre-exists before installing — helm --create-namespace only creates the release namespace.
      If a parent chart grants RBAC to this ServiceAccount, its RoleBinding subject namespace
      must also be "{{ include "orchestrator.namespace" . }}".
{{- end }}
